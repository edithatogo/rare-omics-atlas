  -o "$work/esummary.json.part"
python - "$work/esummary.json.part" <<'PY'
import hashlib
import json
import sys
from pathlib import Path
p = Path(sys.argv[1])
b = p.read_bytes()
r = json.loads(b)["result"]
expected = {"9887916", "12492390", "23508658"}
assert set(r["uids"]) == expected
assert all(str(r[i]["uid"]) == i and r[i]["title"] for i in expected)
print(json.dumps({"ids": sorted(expected), "size_bytes": len(b), "sha256": hashlib.sha256(b).hexdigest()}))
PY
mv "$work/esummary.json.part" "$work/esummary.json"
